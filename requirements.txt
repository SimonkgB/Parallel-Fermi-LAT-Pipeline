# Runtime dependencies of the pipeline.
#
# All pure Python; install them into the fermitools conda/micromamba env, e.g.
#     micromamba install -n fermi -c conda-forge omegaconf astropy numpy
# and run the pipeline with that env activated (micromamba activate fermi).
#
# The Fermi Science Tools themselves (gtselect, gtmktime, gtbin, gtltcube,
# gtltsum, gtexpcube2) are NOT pip-installable and come from that env.

omegaconf       # ${...} interpolation in configs/*.yaml
astropy         # FITS I/O for counts summing, GTI handling, spacecraft merging/slicing
numpy           # map accumulation and GTI arrays
